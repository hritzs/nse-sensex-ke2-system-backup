#!/usr/bin/env python3
"""
Deterministically rebuild all RUNNING fixed lanes from persisted minute data.

- Uses market_minute_history as authoritative first-tick data.
- Replays every stored minute through the real MinuteSimulation.process_snapshot().
- Evaluates hedge/risk at HH:MM:59 so the production :59 gate is unchanged.
- Preserves first-tick timestamps in P&L history.
- Keeps every hedge event generated by the real engine; no hedge is inserted manually.
- Replays through the latest persisted minute, so later hedges are included.
"""
from __future__ import annotations

import shutil
from datetime import datetime
from pathlib import Path

import simulation_server_2 as mod

ROOT = Path(__file__).resolve().parent
DB = mod.DB_PATH
BACKUP = DB.with_name(DB.name + ".before_full_lane_replay_" + datetime.now().strftime("%Y%m%d_%H%M%S") + ".bak")

print("=" * 72)
print("FULL FIXED-LANE HISTORICAL REPLAY")
print("=" * 72)
print(f"DB     : {DB}")
print(f"BACKUP : {BACKUP}")

if DB.exists():
    shutil.copy2(DB, BACKUP)
    print("[BACKUP] created")

controller = mod.CONTROLLER
original_position_quotes = controller.book.position_quotes


def replay_lane(view: str, run_id: str):
    sim = controller.runs_by_view[view][run_id]
    if not sim.started or sim.completed or not sim.entry_expiry:
        print(f"[SKIP] {view}/{run_id}: status={sim.status} expiry={sim.entry_expiry}")
        return sim

    expiry = sim.entry_expiry
    rows = [r for r in controller.book.session_history(expiry) if isinstance(r, dict)]
    rows.sort(key=lambda r: mod.parse_timestamp(r.get("timestamp")) or datetime.min.replace(tzinfo=mod.IST))
    entry_dt = mod.parse_timestamp(sim.entry_timestamp)
    if entry_dt is not None:
        start = entry_dt.replace(second=0, microsecond=0)
        rows = [r for r in rows if (mod.parse_timestamp(r.get("timestamp")) or entry_dt) >= start]

    if not rows:
        print(f"[SKIP] {view}/{run_id}: no stored market rows for {expiry}")
        return sim

    rebuilt = mod.MinuteSimulation(
        run_id=run_id,
        book=controller.book,
        persistence=controller.persistence,
        trade_date=controller.trade_date,
        expiry_view=view,
    )
    rebuilt.request_timestamp = sim.request_timestamp
    rebuilt.start_timestamp = sim.start_timestamp
    rebuilt.armed = sim.armed

    def historical_position_quotes(expiry_name: str, ce_symbol: str, pe_symbol: str):
        strike = int(round(float(rebuilt.entry_atm or 0)))
        with controller.book.lock:
            return (
                controller.book.options.get((expiry_name, strike, "CE")),
                controller.book.options.get((expiry_name, strike, "PE")),
            )

    def snapshot_from_row(row):
        expiry_name = str(row.get("expiry") or expiry).upper()
        spot = mod.finite_float(row.get("synthetic_future", row.get("spot")))
        atm = mod.finite_float(row.get("atm"))
        ce_px = mod.finite_float(row.get("ce"))
        pe_px = mod.finite_float(row.get("pe"))
        ts = str(row.get("timestamp") or "")
        if not expiry_name or spot is None or atm is None or ce_px is None or pe_px is None:
            return None
        if spot <= 0 or ce_px <= 0 or pe_px <= 0:
            return None

        ce_symbol = rebuilt.entry_ce_symbol or f"NIFTY{expiry_name.replace('-', '')}{int(round(atm))}CE"
        pe_symbol = rebuilt.entry_pe_symbol or f"NIFTY{expiry_name.replace('-', '')}{int(round(atm))}PE"
        ce = mod.OptionQuote(
            symbol=ce_symbol, expiry=expiry_name, strike=float(atm), option_type="CE",
            ltp=ce_px, bid=ce_px, ask=ce_px, timestamp=ts,
        )
        pe = mod.OptionQuote(
            symbol=pe_symbol, expiry=expiry_name, strike=float(atm), option_type="PE",
            ltp=pe_px, bid=pe_px, ask=pe_px, timestamp=ts,
        )
        return {
            "expiry": expiry_name,
            "spot": float(spot),
            "synthetic_future": float(spot),
            "atm": float(atm),
            "ce": ce,
            "pe": pe,
            "ce_ltp": float(ce_px),
            "pe_ltp": float(pe_px),
            "straddle": float(ce_px + pe_px),
            "receive_time": ts,
        }

    try:
        controller.book.position_quotes = historical_position_quotes

        for i, row in enumerate(rows):
            dt = mod.parse_timestamp(row.get("timestamp"))
            if dt is None:
                continue
            snap = snapshot_from_row(row)
            if snap is None:
                continue

            strike = int(round(float(snap["atm"])))
            with controller.book.lock:
                controller.book.options[(expiry, strike, "CE")] = snap["ce"]
                controller.book.options[(expiry, strike, "PE")] = snap["pe"]
                controller.book.synthetic_by_expiry[expiry] = float(snap["spot"])
                controller.book.synthetic_rx_by_expiry[expiry] = mod.monotonic_time.monotonic()

            # The Trading Terminal timestamp carried by this snapshot is authoritative.
            # process_snapshot() stores it as MinuteSimulation._event_time.
            # Do NOT shift it to :59 and do NOT use mod.now_ist().
            before = len(rebuilt.hedge_events)
            rebuilt.process_snapshot(snap)

            # P&L history and hedge events use the same authoritative market timestamp.
            if rebuilt.history:
                rebuilt.history[0]["timestamp"] = dt.isoformat(timespec="seconds")

            if len(rebuilt.hedge_events) > before:
                e = rebuilt.hedge_events[0]
                print(
                    f"[HEDGE] {view}/{run_id} {e.get('timestamp')} "
                    f"old={e.get('old_hedge'):.6f} new={e.get('new_hedge'):.6f} "
                    f"PO={e.get('points_out'):.3f} ALLOW={e.get('points_allowed'):.3f} "
                    f"synthetic={e.get('synthetic'):.2f}"
                )

        # Preserve lifecycle state from the live DB record.
        rebuilt.armed = sim.armed
        rebuilt.completed = sim.completed
        rebuilt.exit_reason = sim.exit_reason
        rebuilt.exit_timestamp = sim.exit_timestamp
        controller.persistence.save_run(rebuilt)
        controller.runs_by_view[view][run_id] = rebuilt

        print(
            f"[DONE] {view}/{run_id} expiry={expiry} "
            f"history={len(rebuilt.history)} hedges={len(rebuilt.hedge_events)} "
            f"latest={rebuilt.history[0]['timestamp'] if rebuilt.history else None} "
            f"pnl={rebuilt.total_pnl:.4f}"
        )
        return rebuilt
    finally:
        controller.book.position_quotes = original_position_quotes


# Replay only lanes that actually belong to the current weekly expiry view.
# Legacy NEXT records pointing at 27-OCT-26 are not replayed.
for view in ("CURRENT", "NEXT"):
    for run_id in mod.RUN_STARTS:
        sim = controller.runs_by_view[view][run_id]
        if view == "NEXT":
            expected = controller.book.expiry_for_role("NEXT")
            if expected is None or sim.entry_expiry != expected:
                print(f"[SKIP INVALID NEXT] {run_id}: stored={sim.entry_expiry} expected={expected}")
                continue
        replay_lane(view, run_id)

print("=" * 72)
print("REPLAY COMPLETE")
print("=" * 72)
print("The DB now contains the full persisted minute history through the latest stored minute.")
print("All hedge events shown above were generated by MinuteSimulation.process_snapshot().")
print(f"Backup: {BACKUP}")
