cmake_minimum_required(VERSION 3.16)
project(ke2_feed_sdk VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_path(LZO_INCLUDE_DIR lzo/lzo1z.h HINTS "${LZO_ROOT}/include")
find_library(LZO_LIBRARY NAMES lzo2 HINTS "${LZO_ROOT}/lib")
if(NOT LZO_INCLUDE_DIR OR NOT LZO_LIBRARY)
  message(FATAL_ERROR "LZO2 development library required; set LZO_ROOT")
endif()
add_library(ke2_feed SHARED src/feed_sdk.cpp ../src/decoder.cpp
  ../src/parser_7208.cpp ../src/market_feed.cpp)
target_include_directories(ke2_feed PUBLIC include PRIVATE ../include "${LZO_INCLUDE_DIR}")
target_compile_definitions(ke2_feed PRIVATE KE2_FEED_BUILD KE2_FEED_LIBRARY)
target_link_libraries(ke2_feed PRIVATE "${LZO_LIBRARY}" Threads::Threads)
set_target_properties(ke2_feed PROPERTIES CXX_VISIBILITY_PRESET hidden
 VISIBILITY_INLINES_HIDDEN YES VERSION 1.0.0 SOVERSION 1)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(ke2_feed PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/cmake/exports.map")
endif()
if(WIN32)
  set_target_properties(ke2_feed PROPERTIES PREFIX "")
  target_link_libraries(ke2_feed PRIVATE ws2_32)
  if(MINGW)
    target_link_options(ke2_feed PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
add_executable(ke2_feed_smoke tests/sdk_smoke.c)
target_link_libraries(ke2_feed_smoke PRIVATE ke2_feed)
if(WIN32)
 target_link_libraries(ke2_feed_smoke PRIVATE ws2_32)
endif()
set_target_properties(ke2_feed_smoke PROPERTIES BUILD_RPATH_USE_ORIGIN TRUE)
if(MINGW)
 target_link_options(ke2_feed_smoke PRIVATE -static -static-libgcc)
endif()
enable_testing()
add_test(NAME sdk_smoke COMMAND ke2_feed_smoke)
install(TARGETS ke2_feed ke2_feed_smoke RUNTIME DESTINATION bin
 LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/ke2_feed.h DESTINATION include)
